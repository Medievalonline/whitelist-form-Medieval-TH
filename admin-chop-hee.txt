  // ใช้กับรายการ "รอตรวจสอบ" ที่ผู้สมัครเคยได้รับความเห็นมาก่อนแล้ว
  function getLatestPreviousReviewer(record, historyList){
    const previousReviewer = String(record.previousReviewedBy || '').trim();
    if(record.previousComment && previousReviewer) return previousReviewer;

    for(let i = historyList.length - 1; i >= 0; i--){
      const reviewer = String((historyList[i] && historyList[i].reviewedBy) || '').trim();
      if(reviewer && reviewer !== '-') return reviewer;
    }

    return String(record.reviewedBy || '').trim();
  }

  // ดึงเฉพาะใบสมัครของ "สถานะเดียว" ผ่าน listByStatus() — Firestore query ที่กรอง
  // status ให้ตั้งแต่ฝั่งเซิร์ฟเวอร์ อ่านเฉพาะเอกสารที่ status ตรงเงื่อนไขเท่านั้น

  // แล้ว merge เข้า allRecords (แทนที่ของเดิมด้วยสถานะเดียวกัน กันข้อมูลซ้ำ/ค้าง)
  async function fetchStatus(status, {silent=false}={}){
    if(statusInFlight === status) return; // กันกดซ้อน
    statusInFlight = status;
    if(!silent){ loadingMsg.style.display = 'block'; }
    try{
      const res = await window.storage.listByStatus(status);
      const fresh = (res && res.records || []).map(r => ({ key: r.key, record: JSON.parse(r.value) }));
      const freshKeys = new Set(fresh.map(x=>x.key));
      // เอา record เก่าของ "สถานะนี้" ที่ตอนนี้ไม่ตรงแล้วออก (เช่นเคยเป็น pending
      // แต่ตอนนี้ถูกเปลี่ยนสถานะไปแล้วจากที่อื่น) ส่วน record ของสถานะอื่นที่โหลด
      // ไว้อยู่แล้วไม่แตะต้อง
      allRecords = allRecords.filter(x => !(x.record.status === status && !freshKeys.has(x.key)));
      const existingKeys = new Set(allRecords.map(x=>x.key));
      fresh.forEach(item=>{
        if(existingKeys.has(item.key)){
          allRecords = allRecords.map(x => x.key === item.key ? item : x);
        }else{
          allRecords.push(item);
        }
      });
      allRecords.sort((a,b)=> new Date(b.record.submittedAt) - new Date(a.record.submittedAt));
      loadedStatuses.add(status);
      renderList();
    }catch(e){
      appList.innerHTML = '<div class="empty-msg">โหลดข้อมูลไม่สำเร็จ กรุณากด "โหลดใหม่"</div>';
    }
    if(!silent){ loadingMsg.style.display = 'none'; }
    statusInFlight = null;
  }

  // ถ้าอยู่ในแท็บ "ทั้งหมด") — ไม่ดึงทั้ง collection เหมือนของเดิม

  // เลยยิงพร้อมกันได้ทั้ง 5 สถานะโดยไม่กระทบเป้าหมายลด reads
  async function refreshCounts(){

      renderCounts();
    }catch(e){ /* ตัวเลขไม่อัปเดตก็ไม่เป็นไร ไม่กระทบการทำงานหลัก */ }
  }

          // (แก้บั๊กเดิม: ก่อนหน้านี้ record.reviewedBy ไม่เคยถูกอัปเดตในหน่วยความจำ
          //  ต้องกด "โหลดใหม่" ก่อนถึงจะเห็นชื่อผู้ตรวจล่าสุด)

          pill.style.background = s.bg;
          pill.style.borderColor = s.bg;

            : '';
        }catch(err){
          saveStatusEl.textContent = 'บันทึกไม่สำเร็จ ลองอีกครั้ง';
        }

      });

    cutoff.setDate(cutoff.getDate() - 90);

    // ดึงเฉพาะใบ "ไม่ผ่านการพิจารณา" สดๆ ก่อนเช็ค — ไม่พึ่งพา allRecords ที่อาจ
    // ยังไม่เคยโหลดสถานะนี้เข้ามาเลย (เช่นแอดมินยังไม่เคยกดแท็บ "ไม่ผ่าน")
    cleanupStatus.textContent = 'กำลังตรวจสอบใบสมัคร Rejected...';
    document.getElementById('cleanupBtn').disabled = true;
    await fetchStatus('rejected', {silent:true});
    document.getElementById('cleanupBtn').disabled = false;

    const targets = allRecords.filter(({record})=>{
      if(record.status !== 'rejected') return false;
      const refDate = record.reviewedAt || record.submittedAt;
      if(!refDate) return false;
      return new Date(refDate) < cutoff;
    });

    if(targets.length === 0){
      cleanupStatus.textContent = 'ไม่มีใบสมัคร Rejected ที่เกิน 90 วัน';
      return;
    }

    const confirmed = confirm(
      'พบใบสมัคร Rejected ที่เกิน 90 วัน จำนวน ' + targets.length + ' ใบ\n' +
      'ยืนยันลบทั้งหมดถาวร? การลบนี้กู้คืนไม่ได้'
    );
    if(!confirmed) return;

    document.getElementById('cleanupBtn').disabled = true;
    let done = 0, failed = 0;
    for(const {key} of targets){
      try{
        await window.storage.delete(key, true);
        done++;
      }catch(e){ failed++; }
    }
    cleanupStatus.textContent = 'ลบสำเร็จ ' + done + ' ใบ' + (failed ? (', ล้มเหลว ' + failed + ' ใบ') : '');
    document.getElementById('cleanupBtn').disabled = false;
    if(done > 0){
      statusCounts.rejected = Math.max(0, (statusCounts.rejected || 0) - done);
      renderCounts();
    }
    await fetchStatus('rejected', {silent:true});
  });

  async function loginWithToken(token){
    const res = await fetch('https://discord.com/api/users/@me', {
      headers: { Authorization: 'Bearer ' + token }
    });
    if(!res.ok) throw new Error('discord api ' + res.status);
    const u = await res.json();

    if(!STAFF_DISCORD_IDS.includes(u.id)){
      clearSession();
      deniedMsg.classList.add('show');
      return;
    }

    staffUser = { id: u.id, username: u.username + (u.discriminator && u.discriminator !== '0' ? '#' + u.discriminator : '') };
    authWrap.style.display = 'none';
    panelWrap.style.display = 'block';
    applicationControlWrap.style.display = 'block';
    document.getElementById('whoAmI').textContent = 'เข้าสู่ระบบในชื่อ: ' + staffUser.username;
    await Promise.all([loadAnswerKeys(), loadApplicationSettings()]);
    fetchStatus('pending'); // โหลดเริ่มต้นเฉพาะ "รอตรวจสอบ" เท่านั้น เพื่อลด reads
    refreshCounts(); // ตัวเลขบนแท็บ — ใช้ aggregation query ต้นทุนต่ำ ไม่กระทบเป้าหมายลด reads
    // หมายเหตุ: เอาการโหลดใหม่อัตโนมัติออกแล้วตามที่ขอ — หน้าเว็บจะไม่รีเฟรชข้อมูลเอง
    // ผู้ใช้ต้องกดปุ่ม "โหลดใหม่" เองเมื่อต้องการข้อมูลล่าสุด
  }

  async function init(){
    const fresh = parseAccessToken();
    if(fresh){
      history.replaceState(null, '', REDIRECT_URI);
      saveSession(fresh.token, fresh.expiresIn);
      try{
        await loginWithToken(fresh.token);
      }catch(e){
        clearSession();
        deniedMsg.textContent = 'เข้าสู่ระบบไม่สำเร็จ กรุณาลองใหม่';
        deniedMsg.classList.add('show');
      }
      return;
    }

    const saved = loadSession();
    if(saved){
      try{
        await loginWithToken(saved);
      }catch(e){
        clearSession();
      }
    }
  }
  init();
})();
